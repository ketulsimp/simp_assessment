-- Tables
CREATE TABLE authors (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    country VARCHAR(50)
);

CREATE TABLE books (
    id SERIAL PRIMARY KEY,
    title VARCHAR(255),
    author_id INT REFERENCES authors(id),
    genre VARCHAR(50),
    price DECIMAL(8,2),
    published_year INT
);

CREATE TABLE members (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100),
    join_date DATE
);

CREATE TABLE loans (
    id SERIAL PRIMARY KEY,
    member_id INT REFERENCES members(id),
    book_id INT REFERENCES books(id),
    loan_date DATE,
    return_date DATE  -- NULL if not yet returned
);

Write SQL queries:

1. Find each author's total book-revenue (sum of prices of books that have been LOANED AND RETURNED). Show author name and total.
2. Find members with currently active loans (return_date IS NULL), along with the count of their active loans.
3. For each genre, find the most expensive book using a window function. Show genre, title, price.




Answer: 

1) 	SELECT 
		a.name,
		SUM(b.price) AS total_revenue
	FROM authors a
	JOIN books b ON a.id = b.author_id
	JOIN loans l ON b.id = l.book_id
	WHERE l.return_date IS NOT NULL
	GROUP BY a.name;

2) select  
m.name, count(l.id) as active_loans 
from members m join loans l on m.id = l.member_id 
where l.return_date is null group by m.name;

3)SELECT genre, title, price
FROM (
    SELECT 
        genre,
        title,
        price,
        ROW_NUMBER() OVER (PARTITION BY genre ORDER BY price DESC) AS rn
    FROM books
) t
WHERE rn = 1;
