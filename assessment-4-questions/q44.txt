-- 1.Find total spending per customer (only delivered orders)
SELECT C.NAME,SUM(OI.QUANTITY*OI.UNIT_PRICE) AS TOTAL_SPENT
FROM CUSTOMERS C 
JOIN ORDERS O ON O.CUSTOMER_ID=C.ID
JOIN ORDER_ITEMS OI ON OI.ORDER_ID=O.ID
WHERE O.STATUS='delivered'
GROUP BY C.NAME;

-- 2. Count number of orders per status.
SELECT STATUS,COUNT(STATUS) AS NO_OF_ORDERS FROM ORDERS GROUP BY STATUS;

-- 3. Find top 2 most expensive products in each category.
WITH RANK_PRODUCTS AS(
	SELECT *,DENSE_RANK() OVER(PARTITION BY CATEGORY ORDER BY PRICE DESC) AS RANK_OF_PRODUCT
FROM PRODUCTS
)SELECT * FROM RANK_PRODUCTS WHERE RANK_OF_PRODUCT<=2;
